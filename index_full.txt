/*
|--------------------------------------------------------------------------
| AJAX
|--------------------------------------------------------------------------
*/

if (isset($_GET['ajax'])) {

    header('Content-Type: application/json; charset=utf-8');
    header('Cache-Control: no-store, no-cache, must-revalidate, max-age=0');

    try {

        /*
        |--------------------------------------------------------------------------
        | AIRPORT SEARCH
        |--------------------------------------------------------------------------
        */

        if ($_GET['ajax'] === 'airport_search') {

            $q = trim($_GET['q'] ?? '');

            if ($q === '') {
                echo json_encode([
                    'success' => true,
                    'airports' => []
                ]);
                exit;
            }

            $like = '%' . $q . '%';

            $sql = "
                SELECT
                    airport_id,
                    airport_code,
                    airport_name,
                    city,
                    country,
                    icao_code
                FROM airport
                WHERE
                    airport_code LIKE :code
                    OR airport_name LIKE :name
                    OR city LIKE :city
                    OR country LIKE :country
                    OR keywords LIKE :keywords
                    OR icao_code LIKE :icao
                ORDER BY
                    CASE
                        WHEN UPPER(airport_code) = UPPER(:exact_code)
                            THEN 0
                        WHEN UPPER(city) = UPPER(:exact_city)
                            THEN 1
                        WHEN UPPER(city) LIKE UPPER(:city_prefix)
                            THEN 2
                        WHEN UPPER(airport_name) LIKE UPPER(:name_prefix)
                            THEN 3
                        ELSE 4
                    END,
                    airport_name ASC
                LIMIT 10
            ";

            $stmt = $pdo->prepare($sql);

            $stmt->execute([
                ':code' => $like,
                ':name' => $like,
                ':city' => $like,
                ':country' => $like,
                ':keywords' => $like,
                ':icao' => $like,
                ':exact_code' => $q,
                ':exact_city' => $q,
                ':city_prefix' => $q . '%',
                ':name_prefix' => $q . '%'
            ]);

            echo json_encode([
                'success' => true,
                'airports' => $stmt->fetchAll(PDO::FETCH_ASSOC)
            ]);

            exit;
        }


        /*
        |--------------------------------------------------------------------------
        | FLIGHT CALENDAR
        |--------------------------------------------------------------------------
        */

        if ($_GET['ajax'] === 'flight_calendar') {

            $from = strtoupper(trim($_GET['from'] ?? ''));
            $to = strtoupper(trim($_GET['to'] ?? ''));

            $start = trim($_GET['start'] ?? '');
            $end = trim($_GET['end'] ?? '');

            if (
                $from === '' ||
                $to === '' ||
                $from === $to
            ) {
                echo json_encode([
                    'success' => false,
                    'dates' => [],
                    'message' => 'Valid airports are required.'
                ]);
                exit;
            }

            $sql = "
                SELECT
                    DATE(f.departure_time) AS flight_date,
                    MIN(f.fare) AS lowest_fare,
                    COUNT(*) AS flight_count
                FROM flight f
                INNER JOIN airport src
                    ON src.airport_id = f.source_airport_id
                INNER JOIN airport dst
                    ON dst.airport_id = f.destination_airport_id
                WHERE
                    UPPER(src.airport_code) = :from_code
                    AND UPPER(dst.airport_code) = :to_code
            ";

            $params = [
                ':from_code' => $from,
                ':to_code' => $to
            ];

            if (
                preg_match('/^\d{4}-\d{2}-\d{2}$/', $start) &&
                preg_match('/^\d{4}-\d{2}-\d{2}$/', $end)
            ) {

                $sql .= "
                    AND DATE(f.departure_time)
                    BETWEEN :start_date AND :end_date
                ";

                $params[':start_date'] = $start;
                $params[':end_date'] = $end;
            }

            $sql .= "
                GROUP BY DATE(f.departure_time)
                ORDER BY flight_date ASC
            ";

            $stmt = $pdo->prepare($sql);
            $stmt->execute($params);

            $dates = [];

            foreach ($stmt->fetchAll(PDO::FETCH_ASSOC) as $row) {

                $dates[$row['flight_date']] = [
                    'price' => (float)$row['lowest_fare'],
                    'flight_count' => (int)$row['flight_count']
                ];
            }

            echo json_encode([
                'success' => true,
                'from' => $from,
                'to' => $to,
                'dates' => $dates
            ]);

            exit;
        }


        echo json_encode([
            'success' => false,
            'message' => 'Unknown AJAX request.'
        ]);

        exit;

    } catch (Throwable $e) {

        http_response_code(500);

        echo json_encode([
            'success' => false,
            'message' => $e->getMessage(),
            'airports' => [],
            'dates' => []
        ]);

        exit;
    }
}

    function escapeHtml(value) {

        return String(value)
            .replaceAll('&', '&amp;')
            .replaceAll('<', '&lt;')
            .replaceAll('>', '&gt;')
            .replaceAll('"', '&quot;')
            .replaceAll("'", '&#039;');

    }


    function validAirportCode(value) {

    return (
        value &&
        value.trim().length >= 2
    );

}


function airportCode(value) {

    if (!value) {
        return '';
    }

    const text =
        value.trim();


    /*
    |--------------------------------------------------------------------------
    | Direct airport code
    |--------------------------------------------------------------------------
    */

    if (
        /^[A-Z]{3}$/i.test(text)
    ) {

        return text.toUpperCase();

    }


    /*
    |--------------------------------------------------------------------------
    | Airport displayed as:
    |
    | Bengaluru (BLR)
    |--------------------------------------------------------------------------
    */

    const match =
        text.match(
            /\(([A-Z]{3})\)/
        );

    if (match) {

        return match[1]
            .toUpperCase();

    }


    return '';

}





    function todayString() {

        const d = new Date();

        const year =
            d.getFullYear();

        const month =
            String(
                d.getMonth() + 1
            ).padStart(2, '0');

        const day =
            String(
                d.getDate()
            ).padStart(2, '0');

        return (
            year +
            '-' +
            month +
            '-' +
            day
        );

    }

                day: '2-digit',
                month: 'short',
                year: 'numeric'
            }
        );

    }

function setupAutocomplete(
    input,
    dropdown
) {

    let timer = null;


    input.addEventListener(
        'input',
        function () {

            clearTimeout(timer);


            /*
            |--------------------------------------------------------------------------
            | User changed the text
            |--------------------------------------------------------------------------
            */

            input.dataset.airportId = '';
            input.dataset.code = '';
            input.dataset.city = '';
            input.dataset.name = '';


            const q =
                input.value.trim();


            if (q.length < 2) {

                dropdown.innerHTML = '';

                dropdown.classList.remove(
                    'open'
                );

                return;

            }


            timer =
                setTimeout(
                    async function () {

                        try {

                            const response =
                                await fetch(
                                    'index.php?ajax=airport_search&q='
                                    +
                                    encodeURIComponent(q),
                                    {
                                        method: 'GET',
                                        cache: 'no-store'
                                    }
                                );


                            if (!response.ok) {

                                throw new Error(
                                    'Airport search failed: HTTP '
                                    +
                                    response.status
                                );

                            }


                            const data =
                                await response.json();


                            dropdown.innerHTML = '';


                            if (
                                !data.success ||
                                !data.airports ||
                                data.airports.length === 0
                            ) {

                                dropdown.innerHTML =
                                    '<div class="airport-option">'
                                    +
                                    '<div class="airport-name">'
                                    +
                                    'No airport found'
                                    +
                                    '</div>'
                                    +
                                    '</div>';

                                dropdown.classList.add(
                                    'open'
                                );

                                return;

                            }


                            data.airports.forEach(
                                function (airport) {

                                    const item =
                                        document.createElement(
                                            'div'
                                        );


                                    item.className =
                                        'airport-option';


                                    /*
                                    |--------------------------------------------------------------------------
                                    | Display
                                    |--------------------------------------------------------------------------
                                    |
                                    | Example:
                                    |
                                    | BLR   Kempegowda International Airport Bengaluru
                                    |       Bengaluru, IN

                                        '<div class="airport-main">'
                                        +

                                        '<span class="airport-code">'

                                        escapeHtml(
                                            airport.airport_code
                                        )
                                        +
                                        '</span>'

                                        +

                                        '<span class="airport-name">'
                                        +
                                        escapeHtml(

                                        +
                                        '</span>'

                                        +

                                        '</div>'

                                        +

                                        '<div class="airport-sub">'

                                            airport.city
                                        )
                                        +
                                        ', '
                                        +
                                        escapeHtml(
                                            airport.country
                                        )
                                        +
                                        '</div>';


                                    /*
                                    |--------------------------------------------------------------------------
                                    | SELECT AIRPORT
                                    |--------------------------------------------------------------------------
                                    */

                                        'click',

                                            /*

                                            | Display city + code

            year
            +
            '-'
            +
            month;


        /*
        |--------------------------------------------------------------------------
        | Calendar API
        |--------------------------------------------------------------------------
        */

        const url =
            'api/calendar.php'
            +
            '?source='
            +
            encodeURIComponent(
                fromCode
            )
            +
            '&destination='
            +
            encodeURIComponent(
                toCode
            )
            +
            '&month='
            +
            encodeURIComponent(
                monthText
            );


        console.log(
            'Calendar URL:',
            url
        );


        const response =
            await fetch(
                url,
                {
                    method: 'GET',
                    cache: 'no-store'
                }
            );


        if (!response.ok) {

            throw new Error(
                'Calendar API error: HTTP '
                +
                response.status
            );

        }


        const data =
            await response.json();


        console.log(
            'Calendar response:',
            data
        );


        if (
            !data.success
        ) {

            calendarDates = {};

            console.warn(
                data.message ||
                'No flight dates found.'
            );

            return;

        }


        calendarDates = {};


        /*
        |--------------------------------------------------------------------------
        | API returns:
        |
        | dates: [
        |     {
        |         date: "2026-08-26",
        |         lowest_fare: 4250,
        |         flight_count: 1
        |     }
        | ]
        |--------------------------------------------------------------------------
        */

        if (
            Array.isArray(
                data.dates
            )
        ) {

            data.dates.forEach(
                function (item) {

                    calendarDates[
                        item.date
                    ] = {

                        lowest_fare:
                            Number(
                                item.lowest_fare
                            ),

                        flight_count:
                            Number(
                                item.flight_count
                            )

                    };

                }
            );

        }


    } catch (error) {

        console.error(
            'Calendar loading failed:',
            error
        );

        calendarDates = {};

    }

}


/*
|--------------------------------------------------------------------------
| BUILD MONTH
|--------------------------------------------------------------------------
*/

function buildMonth(
    year,
    month
) {

    const firstDay =
        new Date(
            year,
            month,
            1
        );


    const lastDay =
        new Date(
            year,
            month + 1,
            0
        );


    const monthName =
        firstDay.toLocaleDateString(
            'en-IN',
            {
                month: 'long',
                year: 'numeric'
            }
        );


    const firstWeekday =
        firstDay.getDay();


    const daysInMonth =
        lastDay.getDate();


    let html = '';


    html +=
        '<div class="month">';


    /*
    |--------------------------------------------------------------------------
    | MONTH HEADER
    |--------------------------------------------------------------------------
    */

    html +=
        '<div class="month-title">'
        +
        escapeHtml(
            monthName
        )
        +
        '</div>';


    /*
    |--------------------------------------------------------------------------
    | WEEKDAYS
    |--------------------------------------------------------------------------
    */

    html +=
        '<div class="weekdays">';


    const weekdays = [

        'Sun',
        'Mon',
        'Tue',
        'Wed',
        'Thu',
        'Fri',
        'Sat'

    ];


    weekdays.forEach(
        function(day) {

            html +=
                '<div class="weekday">'
                +
                day
                +
                '</div>';

        }
    );


    html +=
        '</div>';


    /*
    |--------------------------------------------------------------------------
    | DAYS
    |--------------------------------------------------------------------------
    */

    html +=
        '<div class="days">';


    /*
    |--------------------------------------------------------------------------
    | EMPTY CELLS BEFORE FIRST DAY
    |--------------------------------------------------------------------------
    */

    for (
        let i = 0;
        i < firstWeekday;
        i++
    ) {

        html +=
            '<div class="day empty"></div>';

    }


    /*
    |--------------------------------------------------------------------------
    | TODAY
    |--------------------------------------------------------------------------
    */

    const today =
        todayString();


    /*
    |--------------------------------------------------------------------------
    | CREATE DAYS
    |--------------------------------------------------------------------------
    */

    for (
        let day = 1;
        day <= daysInMonth;
        day++
    ) {


        const date =
            year
            +
            '-'
            +
            String(
                month + 1
            ).padStart(
                2,
                '0'
            )
            +
            '-'
            +
            String(day).padStart(
                2,
                '0'
            );


        const info =
            calendarDates[
                date
            ];


        const isPast =
            date < today;


        const hasFlight =
            !!info;


        const disabled =
            isPast ||
            !hasFlight;


        let classes =
            'day';


        if (isPast) {

            classes +=
                ' past';

        }


        if (!hasFlight) {

            classes +=
                ' unavailable';

        }


        if (
            date ===
            departureDate.value
        ) {

            classes +=
                ' selected';

        }


        if (
            date ===
            returnDate.value
        ) {

            classes +=
                ' selected';

        }


        /*
        |--------------------------------------------------------------------------
        | DAY BUTTON
        |--------------------------------------------------------------------------
        */

        html +=
            '<button'
            +
            ' type="button"'
            +
            ' class="'
            +
            classes
            +
            '"'
            +
            ' data-date="'
            +
            escapeHtml(
                date
            )
            +
            '"'
            +
            (
                disabled
                    ? ' disabled'
                    : ''
            )
            +
            '>';


        html +=
            '<span class="day-number">'
            +
            day
            +
            '</span>';


        /*
        |--------------------------------------------------------------------------
        | PRICE
        |--------------------------------------------------------------------------
        */

        if (
            hasFlight &&
            !isPast
        ) {

            html +=
                '<span class="day-price">'
                +
                formatPrice(
                    info.lowest_fare
                )
                +
                '</span>';


            /*
            |--------------------------------------------------------------------------
            | FLIGHT COUNT
            |--------------------------------------------------------------------------
            */

            html +=
                '<span class="day-flights">'
                +
                info.flight_count
                +
                (
                    info.flight_count === 1
                        ? ' flight'
                        : ' flights'
                )
                +
                '</span>';

        } else {

            html +=
                '<span class="day-no-flight">'
                +
                'No flights'
                +
                '</span>';

        }


        html +=
            '</button>';

    }


    html +=
        '</div>';


    html +=
        '</div>';


    return html;

}


/*
|--------------------------------------------------------------------------
| RENDER CALENDAR
|--------------------------------------------------------------------------
*/

async function renderCalendar() {

    /*
    |--------------------------------------------------------------------------
    | Load API data
    |--------------------------------------------------------------------------
    */

    await loadCalendarDates();


    const year =
        calendarMonth.getFullYear();

    const month =
        calendarMonth.getMonth();


    /*
    |--------------------------------------------------------------------------
    | NEXT MONTH
    |--------------------------------------------------------------------------
    */

    const secondMonth =
        new Date(
            year,
            month + 1,
            1
        );


    /*
    |--------------------------------------------------------------------------
    | Build both months
    |--------------------------------------------------------------------------
    */

    calendarBody.innerHTML =

        '<div class="months">'
        +

        buildMonth(
            year,
            month
        )

        +

        buildMonth(
            secondMonth.getFullYear(),
            secondMonth.getMonth()
        )

        +

        '</div>';


    /*
    |--------------------------------------------------------------------------
    | DATE CLICK
    |--------------------------------------------------------------------------
    */

    calendarBody
        .querySelectorAll(
            '.day[data-date]'
        )
        .forEach(
            function(button) {

                button.addEventListener(
                    'click',
                    function() {

                        if (
                            button.disabled
                        ) {

                            return;

                        }


                        selectCalendarDate(
                            button.dataset.date
                        );

                    }
                );

            }
        );

}


/*
|--------------------------------------------------------------------------
| OPEN CALENDAR
|--------------------------------------------------------------------------
*/

async function openCalendar(
    mode
) {

    calendarMode =
        mode;


    /*
    |--------------------------------------------------------------------------
    | Open
    |--------------------------------------------------------------------------
    */

    calendar.classList.add(
        'open'
    );


    const from =
        fromAirport.value.trim();


    const to =
        toAirport.value.trim();


    /*
    |--------------------------------------------------------------------------
    | Calendar route
    |--------------------------------------------------------------------------
    */

    if (
        from &&
        to
    ) {

        calendarRoute.textContent =

            from
            +
            ' → '
            +
            to
            +
            ' · '
            +
            (
                mode === 'departure'
                    ? 'Select departure date'
                    : 'Select return date'
            );

    } else {

        calendarRoute.textContent =
            'Select airports first';

    }


    /*
    |--------------------------------------------------------------------------
    | Render
    |--------------------------------------------------------------------------
    */

    await renderCalendar();

}


/*
|--------------------------------------------------------------------------
| DATE SELECTION
|--------------------------------------------------------------------------
*/

function selectCalendarDate(
    value
) {


    /*
    |--------------------------------------------------------------------------
    | DEPARTURE
    |--------------------------------------------------------------------------
    */

    if (
        calendarMode ===
        'departure'
    ) {

        departureDate.value =
            value;


        departurePicker.value =
            displayDate(
                value
            );


        /*
        |--------------------------------------------------------------------------
        | ROUND TRIP
        |--------------------------------------------------------------------------
        */

        if (
            tripType.value ===
            'round'
        ) {

            calendarMode =
                'return';


            calendarRoute.textContent =

                fromAirport.value
                +
                ' → '
                +
                toAirport.value
                +
                ' · Select return date';


            /*
            | Render again so the selected
            | departure date is visible.
            */

            renderCalendar();


            return;

        }


        /*
        |--------------------------------------------------------------------------
        | ONE WAY
        |--------------------------------------------------------------------------
        */

        calendar.classList.remove(
            'open'
        );


        return;

    }


    /*
    |--------------------------------------------------------------------------
    | RETURN DATE
    |--------------------------------------------------------------------------
    */

    if (
        !departureDate.value
    ) {

        alert(
            'Please select a departure date first.'
        );

        return;

    }


    if (
        value <=
        departureDate.value
    ) {

        alert(
            'Return date must be after departure date.'
        );

        return;

    }


    returnDate.value =
        value;


    returnPicker.value =
        displayDate(
            value
        );


    calendar.classList.remove(
        'open'
    );

}
